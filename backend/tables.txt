create table tipos_limpeza (

  id_limpeza BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  nome varchar (100) not null unique,
  preco_por_m2 decimal (10, 2) not null,
  descricao text

);


create table pedidos_orcamento (

  id_pedidos BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  nome_cliente varchar (150) not null,
  telefone varchar (20) not null, 
  email varchar (150) not null,
  metragem decimal (10, 2) not null,
  tipo_de_limpeza int,
  valor_total decimal (10, 2) not null,
  status varchar (20) default 'pendente',
  criado_em timestamp default current_timestamp,
  foreign key (tipo_de_limpeza) references tipos_limpeza (id_limpeza) on delete set null
);
